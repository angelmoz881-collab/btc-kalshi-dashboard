y(model.get("current")), f"{pyth.get('age_sec', 0):.1f}s old")

    q1, q2, q3, q4 = st.columns(4)

        count, unit_cost = _contracts_for_trade(side, snapshot, settings["max_contracts"], settings["max_trade_cost"])
