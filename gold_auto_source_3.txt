def _quote_cents(market, side, kind):
    for key in (f"{side}_{kind}_dollars", f"{side}_{kind}"):
        value = market.get(key) if market else None
        if value is None or value == "":
            continue
        try:
            x = float(value)
            return x * 100 if key.endswith("_dollars") else x
        except (TypeError, ValueError):
            continue
    return None


def _book_levels(book, side):
    if not isinstance(book, dict):
        return []
    book = book.get("orderbook_fp") or book
    dollar_key = f"{side}_dollars"
    if dollar_key in book:
        rows, dollar_prices = book.get(dollar_key) or [], True
    else:
        rows, dollar_prices = book.get(side) or [], False
    out = []
    for row in rows:
        try:
            if isinstance(row, dict):
                p = row.get("price_dollars", row.get("price"))
                q = row.get("count_fp", row.get("count", row.get("quantity", row.get("size", 0))))
                is_dollars = dollar_prices or row.get("price_dollars") is not None
            else:
                p, q = row[:2]
                is_dollars = dollar_prices
            price = float(p) * (100 if is_dollars else 1)
            qty = float(q)
            if 0 <= price <= 100 and qty > 0:
                out.append((price, qty))
        except Exception:
            continue
    return sorted(out, key=lambda x: x[0], reverse=True)


def market_snapshot(ticker, mode="PAPER"):
    market = fetch_market(ticker, mode)
    book = fetch_orderbook(ticker, mode)
    yes = _book_levels(book, "yes")
    no = _book_levels(book, "no")
    yes_bid = _quote_cents(market, "yes", "bid")
    no_bid = _quote_cents(market, "no", "bid")
    if yes_bid is None and yes:
        yes_bid = yes[0][0]
    if no_bid is None and no:
        no_bid = no[0][0]
    yes_ask = _quote_cents(market, "yes", "ask")
    no_ask = _quote_cents(market, "no", "ask")
    if yes_ask is None and no_bid is not None:
        yes_ask = 100 - no_bid
    if no_ask is None and yes_bid is not None:
        no_ask = 100 - yes_bid
    return {
        "market": market,
        "yes": yes,
        "no": no,
        "yes_bid": yes_bid,
        "no_bid": no_bid,
        "yes_ask": yes_ask,
        "no_ask": no_ask,
    }


def extract_strike(market):
    if not market:
        return None
    for key in ("floor_strike", "cap_strike", "strike_price"):
        value = _safe_float(market.get(key))
        if value is not None and 100 < value < 100000:
            return value
    import re
    text = " ".join(str(market.get(k, "")) for k in ("title", "subtitle", "rules_primary", "rules_secondary"))
    matches = re.findall(r"\$?([0-9][0-9,]*\.?[0-9]*)", text)
    for token in reversed(matches):
        try:
            value = float(token.replace(",", ""))
            if 100 < value < 100000:
                return value
        except ValueError:
            pass
    return None


def _rsi(close, period=14):
    c = pd.Series(close, dtype="float64")
    d = c.diff()
    gain = d.clip(lower=0).ewm(alpha=1 / period, adjust=False).mean()
    loss = (-d.clip(upper=0)).ewm(alpha=1 / period, adjust=False).mean()
    rs = gain / loss.replace(0, np.nan)
    return (100 - 100 / (1 + rs)).fillna(50.0)


def _fee_estimate(price_dollars):
    """Conservative per-contract taker fee approximation; verify actual Kalshi fee schedule/order ticket."""
    p = float(np.clip(price_dollars, 0.001, 0.999))
    raw = 0.07 * p * (1.0 - p)
    return math.ceil(raw * 100.0 - 1e-12) / 100.0

    expiry = _to_utc(market.get("close_time") or market.get("expiration_time"))
    now = pd.Timestamp.now(tz="UTC")
    remaining_sec = max(0.0, (expiry - now).total_seconds()) if expiry is not None else 900.0
    remaining_min = max(remaining_sec / 60.0, 0.25)

        "reversal_risk": "—",
        "yes_ev": None,
        "no_ev": None,
        "yes_fee": None,
        "no_fee": None,
        "spread": None,
        "market_p": None,
        "stat_p": None,
        "momentum_p": None,
        "sigma_1m": None,
        "z_distance": None,
        "rsi": None,
    }
    if strike is None or candles is None or len(candles) < 35:
        return out

    df = candles.copy()
    for col in ("open", "high", "low", "close", "volume"):
        df[col] = pd.to_numeric(df[col], errors="coerce")

    basis = pyth_price / gc_last
