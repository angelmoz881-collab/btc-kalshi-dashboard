def _secret(name, default=None):
    try:
        value = st.secrets.get(name, default)
    except Exception:
        value = os.environ.get(name, default)
    return value


def _bool_secret(name, default=False):
    value = _secret(name, default)
    if isinstance(value, bool):
        return value
    return str(value).strip().lower() in {"1", "true", "yes", "on"}


def _safe_float(value):
    try:
        x = float(value)
        return x if math.isfinite(x) else None
    except (TypeError, ValueError):
        return None


def _to_utc(value):
    if not value:
        return None
    try:
        return pd.to_datetime(value, utc=True)
    except Exception:
        return None


def _normal_cdf(x):
    return 0.5 * (1.0 + math.erf(x / math.sqrt(2.0)))


def _logistic(x):
    return 1.0 / (1.0 + math.exp(-float(np.clip(x, -12, 12))))

def _load_private_key(private_pem):
    if not private_pem:
        raise ValueError("Missing Kalshi private key")
    try:
        from cryptography.hazmat.primitives import serialization
    except ImportError as exc:
        raise RuntimeError("Install cryptography to enable authenticated Kalshi trading") from exc
    return serialization.load_pem_private_key(str(private_pem).encode("utf-8"), password=None)


def _sign_request(private_key, timestamp_ms, method, path):
    message = f"{timestamp_ms}{method.upper()}{path.split('?')[0]}".encode("utf-8")
    try:
        from cryptography.hazmat.primitives import hashes
        from cryptography.hazmat.primitives.asymmetric import padding
        from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PrivateKey
    except ImportError as exc:
        raise RuntimeError("Install cryptography to enable authenticated Kalshi trading") from exc

    if isinstance(private_key, Ed25519PrivateKey):
        signature = private_key.sign(message)
    else:
        signature = private_key.sign(
            message,
            padding.PSS(mgf=padding.MGF1(hashes.SHA256()), salt_length=padding.PSS.DIGEST_LENGTH),
            hashes.SHA256(),
        )
    return base64.b64encode(signature).decode("utf-8")


def _auth_headers(mode, method, path):
    key_id, private_pem = _trade_credentials(mode)
    if not key_id or not private_pem:
        raise ValueError("Kalshi API credentials are not configured")
    private_key = _load_private_key(private_pem)
    ts = str(int(time.time() * 1000))
    signature = _sign_request(private_key, ts, method, path)
    return {
