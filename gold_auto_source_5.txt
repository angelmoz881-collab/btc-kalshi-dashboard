one or ask_cents <= 0:
        return 0, None
    cost = float(ask_cents) / 100.0
    affordable = int(math.floor(float(max_cost) / max(cost, 1e-9)))
    count = max(0, min(int(max_contracts), affordable))
    return count, cost


def _client_order_id(mode, ticker, side):
    # Stable per market+direction, so a Streamlit rerun cannot create another order for the same signal.
    return str(uuid.uuid5(uuid.NAMESPACE_URL, f"kalshi-gold-auto:{mode}:{ticker}:{side}"))


def place_order(mode, ticker, side, count, snapshot):
    if side == "YES":
        yes_book_price = Decimal(str(snapshot["yes_ask"] / 100.0))
        book_side = "bid"
    else:
        # V2 uses a single YES-price book: an ask at p gives NO exposure costing 1-p.
        no_cost = Decimal(str(snapshot["no_ask"] / 100.0))
        yes_book_price = Decimal("1.0") - no_cost
        book_side = "ask"

    price = yes_book_price.quantize(Decimal("0.0001"), rounding=ROUND_DOWN)
    price = min(Decimal("0.9900"), max(Decimal("0.0100"), price))
    payload = {
        "ticker": ticker,
        "side": book_side,
        "count": f"{Decimal(count):.2f}",
        "price": f"{price:.4f}",
        "time_in_force": "immediate_or_cancel",
        "self_trade_prevention_type": "taker_at_cross",
        "client_order_id": _client_order_id(mode, ticker, side),
        "post_only": False,
        "cancel_order_on_pause": True,
        "reduce_only": False,
    }
    result = authenticated_request(mode, "POST", f"{TRADE_PREFIX}/portfolio/events/orders", json_body=payload)
    return payload, result


def _paper_fill(ticker, side, count, snapshot, model):
    ask = snapshot.get("yes_ask") if side == "YES" else snapshot.get("no_ask")
    row = {
        "time_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        "ticker": ticker,
        "side": side,
        "count": count,
        "ask_cents": ask,
        "model_prob_yes": model.get("prob_yes"),
        "confidence": model.get("confidence"),
        "yes_ev": model.get("yes_ev"),
        "no_ev": model.get("no_ev"),
        "client_order_id": _client_order_id("PAPER", ticker, side),
    }
    st.session_state.setdefault("gold_paper_fills", []).append(row)
    return row
